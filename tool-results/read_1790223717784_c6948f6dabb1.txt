    52→    52→function useClickOutside(ref: RefObject<HTMLDivElement | null>, handler: () => void) {
    53→    53→  useEffect(() => {
    54→    54→    const listener = (event: MouseEvent | TouchEvent) => {
    55→    55→      if (!ref.current || ref.current.contains(event.target as Node)) return;
    56→    56→      handler();
    57→    57→    };
    58→    58→    document.addEventListener("mousedown", listener);
    59→    59→    document.addEventListener("touchstart", listener);
    60→    60→    return () => {
    61→    61→      document.removeEventListener("mousedown", listener);
    62→    62→      document.removeEventListener("touchstart", listener);
    63→    63→    };
    64→    64→  }, [ref, handler]);
    65→    65→}

   203→   203→function ExpandableCard({
   204→   204→  card,
   205→   205→  index,
   206→   206→  layout = false,
   207→   207→  idPrefix,
   208→   208→}: {

   210→   210→  index: number;
   211→   211→  layout?: boolean;
   212→   212→  idPrefix: string;
   213→   213→}) {
   214→   214→  const [active, setActive] = useState(false);
   215→   215→  const ref = useRef<HTMLDivElement>(null);
   216→   216→  const { onCardClose } = useContext(CarouselContext);
   217→   217→
   218→   218→  const handleClose = () => {
   219→   219→    setActive(false);
   220→   220→    onCardClose(index);
   221→   221→  };
   222→   222→
   223→   223→  useEffect(() => {
   224→   224→    const onKeyDown = (event: KeyboardEvent) => {
   225→   225→      if (event.key === "Escape") handleClose();

   228→   228→    else document.body.style.overflow = "auto";

   233→   233→  useClickOutside(ref, handleClose);
   234→   234→
   235→   235→  return (
   236→   236→    <>
   237→   237→      <AnimatePresence>
   238→   238→        {active && (
   239→   239→          <div className="fixed inset-0 z-52 h-screen overflow-auto">

   308→   308→/* Horizontal scrolling carousel with arrow controls */
   309→   309→function ExpandableCarousel({ items }: { items: ReactNode[] }) {
   310→   310→  const carouselRef = useRef<HTMLDivElement>(null);
   311→   311→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   312→   312→  const [canScrollRight, setCanScrollRight] = useState(true);
   313→   313→  const [currentIndex, setCurrentIndex] = useState(0);
   314→   314→
   315→   315→  const checkScroll = () => {
   316→   316→    if (carouselRef.current) {
   317→   317→      const { scrollLeft, scrollWidth, clientWidth } = carouselRef.current;
   318→   318→      setCanScrollLeft(scrollLeft > 0);
   319→   319→      setCanScrollRight(scrollLeft < scrollWidth - clientWidth);
   320→   320→    }
   321→   321→  };
   322→   322→
   323→   323→  useEffect(() => {
   324→   324→    if (carouselRef.current) {
   325→   325→      carouselRef.current.scrollLeft = 0;
   326→   326→      checkScroll();
   327→   327→    }
   328→   328→  }, []);
   329→   329→
   330→   330→  const scrollAmount = () => 240;
   331→   331→
   332→   332→  return (
   333→   333→    <CarouselContext.Provider
   334→   334→      value={{
   335→   335→        onCardClose: (index) => {
   336→   336→          if (carouselRef.current) {
   337→   337→            carouselRef.current.scrollTo({ left: (224 + 16) * index, behavior: "smooth" });
   338→   338→            setCurrentIndex(index);
   339→   339→          }
   340→   340→        },
   341→   341→        currentIndex,
   342→   342→      }}
   343→   343→    >
   344→   344→      <div className="relative w-full">
   345→   345→        <div
   346→   346→          className="flex w-full overflow-x-scroll overscroll-x-auto scroll-smooth py-10 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden"
   347→   347→          ref={carouselRef}
   348→   348→          onScroll={checkScroll}
   349→   349→        >
   350→   350→          <div className="absolute right-0 z-[10] h-auto w-[5%] overflow-hidden bg-gradient-to-l" />
   351→   351→          <div className="mx-auto flex max-w-7xl flex-row justify-start gap-4">
   352→   352→            {items.map((item, idx) => (
   353→   353→              <motion.div
   354→   354→                key={"card" + idx}
   355→   355→                initial={{ opacity: 0, y: 20 }}
   356→   356→                animate={{ opacity: 1, y: 0, transition: { duration: 0.5, delay: 0.2 * idx, ease: "easeOut" } }}
   357→   357→                className="rounded-3xl last:pr-[5%] md:last:pr-[33%]"
   358→   358→              >
   359→   359→                {item}
   360→   360→              </motion.div>
   361→   361→            ))}
   362→   362→          </div>
   363→   363→        </div>
   364→   364→        <div className="mr-10 flex justify-end gap-2 md:mr-20">
   365→   365→          <button
   366→   366→            className="relative z-40 flex h-10 w-10 cursor-pointer items-center justify-center rounded-full bg-gray-100 disabled:opacity-50"
   367→   367→            onClick={() => carouselRef.current?.scrollBy({ left: -scrollAmount(), behavior: "smooth" })}
   368→   368→            disabled={!canScrollLeft}
   369→   369→            aria-label="Scroll projects left"
   370→   370→          >
   371→   371→            <IconArrowNarrowLeft className="h-6 w-6 text-gray-500" />
   372→   372→          </button>
   373→   373→          <button
   374→   374→            className="relative z-40 flex h-10 w-10 cursor-pointer items-center justify-center rounded-full bg-gray-100 disabled:opacity-50"
   375→   375→            onClick={() => carouselRef.current?.scrollBy({ left: scrollAmount(), behavior: "smooth" })}
   376→   376→            disabled={!canScrollRight}
   377→   377→            aria-label="Scroll projects right"
   378→   378→          >
   379→   379→            <IconArrowNarrowRight className="h-6 w-6 text-gray-500" />
   380→   380→          </button>
   381→   381→        </div>
   382→   382→      </div>
   383→   383→    </CarouselContext.Provider>
   384→   384→  );
   385→   385→}
