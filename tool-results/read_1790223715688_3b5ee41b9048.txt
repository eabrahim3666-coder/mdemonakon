    52→function useClickOutside(ref: RefObject<HTMLDivElement | null>, handler: () => void) {
    53→  useEffect(() => {
    54→    const listener = (event: MouseEvent | TouchEvent) => {
    55→      if (!ref.current || ref.current.contains(event.target as Node)) return;
    56→      handler();
    57→    };
    58→    document.addEventListener("mousedown", listener);
    59→    document.addEventListener("touchstart", listener);
    60→    return () => {
    61→      document.removeEventListener("mousedown", listener);
    62→      document.removeEventListener("touchstart", listener);
    63→    };
    64→  }, [ref, handler]);
    65→}

   202→/* A single carousel card — click opens the morphing modal overlay */
   203→function ExpandableCard({
   204→  card,
   205→  index,
   206→  layout = false,
   207→  idPrefix,
   208→}: {
   209→  card: ExpandableCardData;
   210→  index: number;
   211→  layout?: boolean;
   212→  idPrefix: string;
   213→}) {
   214→  const [active, setActive] = useState(false);
   215→  const ref = useRef<HTMLDivElement>(null);
   216→  const { onCardClose } = useContext(CarouselContext);
   217→
   218→  const handleClose = () => {
   219→    setActive(false);
   220→    onCardClose(index);
   221→  };
   222→
   223→  useEffect(() => {
   224→    const onKeyDown = (event: KeyboardEvent) => {
   225→      if (event.key === "Escape") handleClose();
   226→    };
   227→    if (active) document.body.style.overflow = "hidden";
   228→    else document.body.style.overflow = "auto";
   229→    window.addEventListener("keydown", onKeyDown);
   230→    return () => window.removeEventListener("keydown", onKeyDown);
   231→  }, [active]);
   232→
   233→  useClickOutside(ref, handleClose);
   234→
   235→  return (
   236→    <>
   237→      <AnimatePresence>
   238→        {active && (
   239→          <div className="fixed inset-0 z-52 h-screen overflow-auto">
   240→            <motion.div
   241→              initial={{ opacity: 0 }}
   242→              animate={{ opacity: 1 }}
   243→              exit={{ opacity: 0 }}
   244→              className="fixed inset-0 h-full w-full bg-black/80 backdrop-blur-lg"
   245→            />
   246→            <motion.div
   247→              initial={{ opacity: 0 }}
   248→              animate={{ opacity: 1 }}
   249→              exit={{ opacity: 0 }}
   250→              ref={ref}
   251→              layoutId={layout ? `${idPrefix}card-${card.title}` : undefined}
   252→              className="relative z-[60] mx-auto my-10 h-fit max-w-5xl rounded-3xl bg-white font-sans dark:bg-neutral-900"
   253→            >
   254→              <div className="sticky top-4 z-52 flex justify-end px-8 pt-8 md:px-14 md:pt-8">
   255→                <button
   256→                  className="flex h-8 w-8 cursor-pointer items-center justify-center rounded-full bg-black/90 shadow-md dark:bg-white/90"
   257→                  onClick={handleClose}
   258→                  aria-label="Close project details"
   259→                >
   260→                  <X className="h-6 w-6 text-neutral-100 dark:text-neutral-900" />
   261→                </button>
   262→              </div>
   263→              <div className="relative px-8 pt-2 pb-0 md:px-14">
   264→                <motion.p
   265→                  layoutId={layout ? `${idPrefix}category-${card.title}` : undefined}
   266→                  className="text-base font-medium text-black dark:text-white"
   267→                >
   268→                  {card.category}
   269→                </motion.p>
   270→                <motion.p
   271→                  layoutId={layout ? `${idPrefix}title-${card.title}` : undefined}
   272→                  className="mt-4 text-2xl font-semibold text-neutral-700 md:text-5xl dark:text-white"
   273→                >
   274→                  {card.title}
   275→                </motion.p>
   276→              </div>
   277→              <div className="px-8 pt-8 pb-14 md:px-14">{card.content}</div>
   278→            </motion.div>
   279→          </div>
   280→        )}
   281→      </AnimatePresence>
   282→      <motion.button
   283→        layoutId={layout ? `${idPrefix}card-${card.title}` : undefined}
   284→        onClick={() => setActive(true)}
   285→        className="relative z-10 flex h-80 w-56 cursor-pointer flex-col items-start justify-start overflow-hidden rounded-3xl bg-gray-100 dark:bg-neutral-900"
   286→      >
   287→        <div className="absolute inset-x-0 top-0 z-30 h-full cursor-pointer bg-gradient-to-b from-black via-transparent to-transparent hover:scale-110" />
   288→        <div className="relative z-40 p-8">
   289→          <motion.p
   290→            layoutId={layout ? `${idPrefix}cat-${card.category}` : undefined}
   291→            className="text-left font-sans text-sm font-medium text-white md:text-base"
   292→          >
   293→            {card.category}
   294→          </motion.p>
   295→          <motion.p
   296→            layoutId={layout ? `${idPrefix}title-${card.title}` : undefined}
   297→            className="max-w-xs text-left font-sans text-xl font-semibold [text-wrap:balance] text-white md:text-3xl"
   298→          >
   299→            {card.title}
   300→          </motion.p>
   301→        </div>
   302→        <BlurImage src={card.src} alt={card.title} className="absolute inset-0 z-10 object-cover" />
   303→      </motion.button>
   304→    </>
   305→  );
   306→}

   308→/* Horizontal scrolling carousel with arrow controls */
   309→function ExpandableCarousel({ items }: { items: ReactNode[] }) {
   310→  const carouselRef = useRef<HTMLDivElement>(null);
   311→  const [canScrollLeft, setCanScrollLeft] = useState(false);
   312→  const [canScrollRight, setCanScrollRight] = useState(true);
   313→  const [currentIndex, setCurrentIndex] = useState(0);
   314→
   315→  const checkScroll = () => {
   316→    if (carouselRef.current) {
   317→      const { scrollLeft, scrollWidth, clientWidth } = carouselRef.current;
   318→      setCanScrollLeft(scrollLeft > 0);
   319→      setCanScrollRight(scrollLeft < scrollWidth - clientWidth);
   320→    }
   321→  };
   322→
   323→  useEffect(() => {
   324→    if (carouselRef.current) {
   325→      carouselRef.current.scrollLeft = 0;
   326→      checkScroll();
   327→    }
   328→  }, []);
   329→
   330→  const scrollAmount = () => 240;
   331→
   332→  return (
   333→    <CarouselContext.Provider
   334→      value={{
   335→        onCardClose: (index) => {
   336→          if (carouselRef.current) {
   337→            carouselRef.current.scrollTo({ left: (224 + 16) * index, behavior: "smooth" });
   338→            setCurrentIndex(index);
   339→          }
   340→        },
   341→        currentIndex,
   342→      }}
   343→    >
   344→      <div className="relative w-full">
   345→        <div
   346→          className="flex w-full overflow-x-scroll overscroll-x-auto scroll-smooth py-10 [scrollbar-width:none] [&::-webkit-scrollbar]:hidden"
   347→          ref={carouselRef}
   348→          onScroll={checkScroll}
   349→        >
   350→          <div className="absolute right-0 z-[10] h-auto w-[5%] overflow-hidden bg-gradient-to-l" />
   351→          <div className="mx-auto flex max-w-7xl flex-row justify-start gap-4">
   352→            {items.map((item, idx) => (
   353→              <motion.div
   354→                key={"card" + idx}
   355→                initial={{ opacity: 0, y: 20 }}
   356→                animate={{ opacity: 1, y: 0, transition: { duration: 0.5, delay: 0.2 * idx, ease: "easeOut" } }}
   357→                className="rounded-3xl last:pr-[5%] md:last:pr-[33%]"
   358→              >
   359→                {item}
   360→              </motion.div>
   361→            ))}
   362→          </div>
   363→        </div>
   364→        <div className="mr-10 flex justify-end gap-2 md:mr-20">
   365→          <button
   366→            className="relative z-40 flex h-10 w-10 cursor-pointer items-center justify-center rounded-full bg-gray-100 disabled:opacity-50"
   367→            onClick={() => carouselRef.current?.scrollBy({ left: -scrollAmount(), behavior: "smooth" })}
   368→            disabled={!canScrollLeft}
   369→            aria-label="Scroll projects left"
   370→          >
   371→            <IconArrowNarrowLeft className="h-6 w-6 text-gray-500" />
   372→          </button>
   373→          <button
   374→            className="relative z-40 flex h-10 w-10 cursor-pointer items-center justify-center rounded-full bg-gray-100 disabled:opacity-50"
   375→            onClick={() => carouselRef.current?.scrollBy({ left: scrollAmount(), behavior: "smooth" })}
   376→            disabled={!canScrollRight}
   377→            aria-label="Scroll projects right"
   378→          >
   379→            <IconArrowNarrowRight className="h-6 w-6 text-gray-500" />
   380→          </button>
   381→        </div>
   382→      </div>
   383→    </CarouselContext.Provider>
   384→  );
   385→}
